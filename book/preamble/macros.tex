\newcommand{\CurrentPart}{}

\newcommand{\gpart}[2]{%
  \clearpage
  \renewcommand{\CurrentPart}{Part #1 \textperiodcentered\ #2}%
  \phantomsection
  \addcontentsline{toc}{chapter}{#1.\space #2}%
  \noindent\colorbox{Ink}{%
    \begin{minipage}{\dimexpr\linewidth-2\fboxsep\relax}
      \vspace{1.5pt}
      {\sffamily\bfseries\large\color{white}%
       \textcolor{Accent!45}{#1}\hspace{0.7em}#2}
      \vspace{1.5pt}
    \end{minipage}}%
  \par\vspace{9pt}%
  \gfreshtrue
}

\newcommand{\gsec}[1]{%
  \ifgfresh\gfreshfalse\else\clearpage\fi
  \section{#1}%
  \markboth{\CurrentPart}{#1}%
}

% A level badge under a piece's title, and the glossary block that sits at the
% foot of the page. The glossary is a dtbl for the same reason the word lists

%
% \tabularx / \endtabularx rather than \begin{tabularx} / \end{tabularx}:
% tabularx scans ahead for its own \end, so splitting it across an
% environment's two halves fails with "File ended while scanning use of
% \TX@get@body". dtbl below is built the same way for the same reason.
\newenvironment{glosstbl}{%
  \par\vspace{3pt}%
  {\color{Line}\hrule height 0.4pt}\vspace{4pt}%
  \footnotesize\color{Faint}%
  \setlength{\extrarowheight}{1.5pt}%
  \setlength{\tabcolsep}{4pt}%
  \tabularx{\linewidth}{@{}W{0.92}W{1.08}SW{0.92}W{1.08}@{}}%
}{%
  \endtabularx
  \par
}

% ---------- test this page -----------------------------------
% One topic per page means the printed page number identifies the topic on
% its own, so the link needs nothing but \thepage. The quiz reads ?page=N,
% finds the section printed on that page and asks only its questions.
%
% \thepage has to be expanded before \href sees it, hence the \edef — \href
% does not expand its first argument.
%
% Each book uses its own query parameter — ?page=N for the grammar book,
% ?story=N for the story book — rather than one parameter plus a book name.
% That avoids putting an & in the URL, which would need catcode surgery to
% survive LaTeX, for no gain.
\newcommand{\TestSiteURL}{https://dmgreenhouse.github.io/German/}
\newcommand{\TestParam}{page}
\newcommand{\testthispage}{%
  \edef\dgtesturl{\TestSiteURL?\TestParam=\thepage}%
  \href{\dgtesturl}{\footnotesize\sffamily\color{Accent}Diese Seite testen
        $\rightarrow$}%
}

% 7. note box
%    Built from a colorbox rather than a table: there are ~90 of these in the
%    book and a tabularray environment for each one dominated the compile time.
\newsavebox{\gnotebox}

\setlength{\gnoterule}{2pt}
\newenvironment{gnote}{%
  \par\vspace{5pt}%
  \setlength{\fboxsep}{5pt}%

  \end{minipage}%
  \end{lrbox}%

  \par\vspace{3pt}%
}

% 8. dtbl — the lightweight table used by the vocabulary and reference pages.
%    Visually it matches gtbl, but it is built on tabularx rather than
%    tabularray. A tabularray environment costs roughly 0.17s to typeset and
%    the book contains hundreds of tables, so the long word lists use this
%    instead and cost essentially nothing.
%
%    Column types: W{w} left-aligned, C{w} centred, weights must sum to the
%    number of columns. Header cells are wrapped in \hd.
%
%    S is a zero-width separator: a vertical rule marking where a table wraps
%    round onto a second set of columns, as the word lists and the alphabet do.
%    It carries no weight, so it does not affect the sum the checker enforces.
\newcolumntype{W}[1]{>{\hsize=#1\hsize\raggedright\arraybackslash}X}
\newcolumntype{C}[1]{>{\hsize=#1\hsize\centering\arraybackslash}X}
\newcolumntype{S}{!{\color{Line}\vrule width 0.5pt}}

\newcommand{\hd}[1]{{\sffamily\bfseries\footnotesize\color{white}#1}}

\newenvironment{dtbl}[2][\small]{%
  \par\vspace{3pt}%
  #1%
  \setlength{\extrarowheight}{2pt}%
  \setlength{\arrayrulewidth}{0.25pt}%
  \arrayrulecolor{Ink}%
  \rowcolors{2}{white}{ZebraBG}%
  \setlength{\tabcolsep}{4pt}%
  \tabularx{\linewidth}{#2}%
  \specialrule{0.9pt}{0pt}{0pt}%
}{%
  \arrayrulecolor{Ink}\specialrule{0.9pt}{0pt}{0pt}%
  \endtabularx
  \arrayrulecolor{black}%
  \par\vspace{2pt}%
}

% closes the header row of a dtbl and switches to hairline rules
\newcommand{\hdrule}{\specialrule{0.7pt}{0pt}{0pt}\arrayrulecolor{Line}}

% one hairline between body rows of a dtbl
\newcommand{\rowrule}{\specialrule{0.25pt}{0pt}{0pt}}
